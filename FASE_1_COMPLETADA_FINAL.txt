💡 EJEMPLOS DE USO
═══════════════════════════════════════════════════════════════════════════════

Validar Teléfono:
  
  import 'validators/validators.dart';
  
  if (Validators.isValidPhone(phone)) {
    print('✅ Válido');
  }

Normalizar Teléfono:
  
  final normalized = Validators.normalizePhoneNumber('(912) 345-6789');
  // "+19123456789"

Guardar de Forma Segura:
  
  import 'services/secure_storage_service.dart';
  
  await SecureStorageService.saveEmergencyContact('Juan', '+1234567890');

Recuperar:
  
  final contact = await SecureStorageService.getEmergencyContact();
  // {'nombre': 'Juan', 'telefono': '+1234567890'}

Limpiar (Logout):
  
  await SecureStorageService.deleteAll();
